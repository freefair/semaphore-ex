// These characterization checks prove the current defects, not fixed behavior.
func captureLog(t *testing.T) *bytes.Buffer {
 var out bytes.Buffer
 old := log.StandardLogger().Out
 log.SetOutput(&out)
 t.Cleanup(func(){log.SetOutput(old)})
 return &out
}
func TestAuditInternalErrorBecomesEmptyBadRequest(t *testing.T) {
 logs := captureLog(t)
 response := httptest.NewRecorder()
 helpers.WriteError(response, errors.New("synthetic database disk full"))
 assert.Equal(t, 400, response.Code)
 assert.Empty(t, response.Body.String())
 assert.Contains(t, logs.String(), "synthetic database disk full")
 t.Logf("observed internal DB failure: status=%d body=%q; cause only in uncorrelated server log", response.Code,response.Body.String())
}
func TestAuditMalformedJSONHasNoFieldDiagnostic(t *testing.T) {
 r := httptest.NewRequest(http.MethodPost,"/test",strings.NewReader(`{"port":"wrong-type"}`))
 response := httptest.NewRecorder()
 var body struct {Port int `json:"port"`}
 assert.False(t,helpers.Bind(response,r,&body))
 assert.Equal(t,400,response.Code)
 assert.Empty(t,response.Body.String())
 t.Logf("observed invalid port type: status=%d body=%q",response.Code,response.Body.String())
}

func TestAuditLDAPClientDiscardsCertificateCause(t *testing.T){
 server:=httptest.NewUnstartedServer(http.HandlerFunc(func(http.ResponseWriter,*http.Request){}))
 server.Config.ErrorLog=stdlog.New(io.Discard,"",0);server.StartTLS();defer server.Close()
 result,err:=identity.NewLDAPClient().Authenticate(context.Background(),pro_interfaces.LDAPClientRequest{
  Configuration:pro_interfaces.LDAPClientConfiguration{ServerURL:strings.Replace(server.URL,"https://","ldaps://",1),TLSMode:pro_interfaces.LDAPTLSModeLDAPS,TrustMode:pro_interfaces.LDAPTrustModeSystem,BindDN:"cn=reader,dc=example,dc=test",BindPassword:"fixture",SearchBaseDN:"dc=example,dc=test",UserFilter:"(uid={{username}})",IdentityAttribute:"entryUUID",UsernameAttribute:"uid",NameAttribute:"cn",EmailAttribute:"mail"},Username:"fixture",Credential:"fixture",
 })
 require.Error(t,err);assert.ErrorIs(t,err,pro_interfaces.ErrLDAPProviderUnavailable);var certificateError x509.UnknownAuthorityError;assert.False(t,errors.As(err,&certificateError));assert.Equal(t,"connection_failed",result.Readiness.Code);assert.Equal(t,"LDAP provider unavailable: connect",err.Error());t.Logf("self-signed TLS failure: error=%q readiness=%s certificate cause unavailable through errors.As",err.Error(),result.Readiness.Code)
}
